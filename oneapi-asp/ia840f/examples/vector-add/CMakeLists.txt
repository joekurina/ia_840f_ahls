cmake_minimum_required(VERSION 3.20)
project(VectorAddAHLS LANGUAGES NONE)
set(AHLS_MEMORY_LIMIT_GIB "55" CACHE STRING "Per-process address-space limit in GiB")
if(NOT AHLS_MEMORY_LIMIT_GIB MATCHES "^[1-9][0-9]*$")
  message(FATAL_ERROR "AHLS_MEMORY_LIMIT_GIB must be a positive integer")
endif()

set(AHLS_ROOT "$ENV{INTELFPGAOCLSDKROOT}" CACHE PATH "AHLS 2026.1 aclsycl installation")
set(QUARTUS_ROOT "" CACHE PATH "Quartus Pro 26.1.1 quartus directory")
set(ASP_ROOT "" CACHE PATH "IA840F board package containing board_env.xml")
set(ASP_VARIANT "ofs_ia840f" CACHE STRING "ofs_ia840f or ofs_ia840f_usm")
set(ASP_FLOW "afu_flat" CACHE STRING "afu_flat or afu_flat_kclk")
set(OPAE_PLATFORM_ROOT "" CACHE PATH "Matching OFS2026.1 PR export")
set(AHLS_REUSE_AOCX "" CACHE FILEPATH "Retained legacy host-link input; use AHLS_REUSE_RTL_ARCHIVE")
set(AHLS_PREBUILT_ARCHIVE "" CACHE FILEPATH "Matching prebuilt SYCL FPGA archive for native image import")
set(AHLS_REUSE_EXECUTABLE "" CACHE FILEPATH "Matching SYCL .fpga executable for native -reuse-exe host relinking")
set(AHLS_REUSE_RTL_ARCHIVE "" CACHE FILEPATH "Matching AHLS early-link archive; reuse unchanged generated kernel")
option(VECTOR_ADD_TRACE_RUNTIME "Record dynamic-loader runtime selection" OFF)
set(ASP_BUILD_CONFIG "" CACHE FILEPATH "Owned configuration for the migrated BSP callback")
set(ASP_RUNTIME_ROOT "${ASP_ROOT}" CACHE PATH "Owned runtime prefix containing activate.sh")
set(VECTOR_ADD_SIZE "10000" CACHE STRING "Full numerical hardware test size")
option(VECTOR_ADD_RUN_WITH_SUDO "Use supported privileged runtime access" OFF)

if(NOT UNIX)
  message(FATAL_ERROR "This AHLS2026.1/OFS2026.1 IA840F build targets Linux")
endif()
if(NOT EXISTS "${AHLS_ROOT}/bin/ahls" OR NOT EXISTS "${AHLS_ROOT}/fpgavars.sh")
  message(FATAL_ERROR "Set AHLS_ROOT to the installed AHLS2026.1 aclsycl directory")
endif()
if(NOT EXISTS "${QUARTUS_ROOT}/bin/quartus_sh")
  message(FATAL_ERROR "Set QUARTUS_ROOT to the selected Quartus26.1.1 quartus directory")
endif()
if(NOT EXISTS "${ASP_ROOT}/board_env.xml")
  message(FATAL_ERROR "Set ASP_ROOT to the selected IA840F board package")
endif()
if(NOT ASP_VARIANT MATCHES "^(ofs_ia840f|ofs_ia840f_usm)$")
  message(FATAL_ERROR "Unsupported ASP_VARIANT")
endif()
if(NOT ASP_FLOW MATCHES "^(afu_flat|afu_flat_kclk)$")
  message(FATAL_ERROR "Unsupported ASP_FLOW")
endif()
find_program(BASH_EXECUTABLE bash REQUIRED)
set(AHLS_COMPILER "${AHLS_ROOT}/bin/ahls")
configure_file(cmake/ahls-env.sh.in ahls-env.sh @ONLY)
configure_file(cmake/runtime-env.sh.in runtime-env.sh @ONLY)
add_subdirectory(src)
