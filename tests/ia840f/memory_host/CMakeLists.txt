cmake_minimum_required(VERSION 3.16)
project(ia840f_memory_host_offline C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(N "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(MEMORY_FRONTEND_SOURCE "${N}/src/host/ahls_memory_inspect.c" CACHE FILEPATH "Actual frontend source")
set(OPAE_CAPTURED_INCLUDE_DIR "" CACHE PATH "Captured installed OPAE headers")
set(TEST_OPTIMIZATION "-O0" CACHE STRING "Fixture optimization")
if(NOT EXISTS "${OPAE_CAPTURED_INCLUDE_DIR}/opae/access.h")
  message(FATAL_ERROR "Captured installed OPAE headers are required")
endif()
add_executable(test_memory_frontend "${MEMORY_FRONTEND_SOURCE}" mock_memory_opae.c "${N}/src/host/ahls_qualification_core.c")
target_include_directories(test_memory_frontend PRIVATE "${OPAE_CAPTURED_INCLUDE_DIR}" "${N}/src/host")
target_compile_options(test_memory_frontend PRIVATE ${TEST_OPTIMIZATION} -Wall -Wextra -Werror -pedantic -UNDEBUG -fsanitize=undefined -fno-sanitize-recover=all)
target_link_options(test_memory_frontend PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME FPGA_Test_memory_frontend_inert COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/test_memory_frontend.py" "$<TARGET_FILE:test_memory_frontend>" "${CMAKE_CURRENT_BINARY_DIR}/test-results.json")
set_tests_properties(FPGA_Test_memory_frontend_inert PROPERTIES TIMEOUT 120)
