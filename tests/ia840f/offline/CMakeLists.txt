cmake_minimum_required(VERSION 3.16)
project(ia840f_ahls_host_offline C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(N "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_library(ahls_qualification_core STATIC "${N}/src/host/ahls_qualification_core.c")
target_include_directories(ahls_qualification_core PUBLIC "${N}/src/host")
target_compile_options(ahls_qualification_core PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(test_ahls_qualification test_ahls_qualification.c)
target_compile_options(test_ahls_qualification PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
target_link_libraries(test_ahls_qualification PRIVATE ahls_qualification_core)
enable_testing()
add_test(NAME FPGA_Test_AHLS_offline_contract COMMAND test_ahls_qualification)
set(OPAE_CAPTURED_INCLUDE_DIR "" CACHE PATH "Captured OPAE headers for inert API-link fixtures")
if(OPAE_CAPTURED_INCLUDE_DIR)
  add_executable(test_opae_frontend "${N}/src/host/ahls_opae_qualification.c" mock_opae.c)
  target_include_directories(test_opae_frontend PRIVATE "${OPAE_CAPTURED_INCLUDE_DIR}")
  target_compile_options(test_opae_frontend PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
  target_link_libraries(test_opae_frontend PRIVATE ahls_qualification_core)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME FPGA_Test_OPAE_inert_API COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/test_opae_frontend.py" "$<TARGET_FILE:test_opae_frontend>")
endif()
set_tests_properties(FPGA_Test_AHLS_offline_contract PROPERTIES TIMEOUT 20)
option(BUILD_OPAE_FRONTEND "Compile OPAE frontend, never run it automatically" OFF)
if(BUILD_OPAE_FRONTEND)
  find_path(OPAE_INCLUDE_DIR opae/access.h REQUIRED)
  find_library(OPAE_LIBRARY opae-c REQUIRED)
  add_executable(ahls_opae_qualification "${N}/src/host/ahls_opae_qualification.c")
  target_include_directories(ahls_opae_qualification PRIVATE "${OPAE_INCLUDE_DIR}")
  target_compile_options(ahls_opae_qualification PRIVATE -Wall -Wextra -Werror -pedantic)
  target_link_libraries(ahls_opae_qualification PRIVATE ahls_qualification_core "${OPAE_LIBRARY}")
endif()
