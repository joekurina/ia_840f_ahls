cmake_minimum_required(VERSION 3.16)
project(ia840f_memory_inspect_native C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_SKIP_RPATH ON)
set(OPAE_LIBRARY "/usr/lib64/libopae-c.so" CACHE FILEPATH "Bound installed core library")
set(OPAE_INCLUDE_DIR "/usr/include" CACHE PATH "Bound installed OPAE headers")
if(NOT EXISTS "${OPAE_LIBRARY}" OR NOT EXISTS "${OPAE_INCLUDE_DIR}/opae/access.h")
  message(FATAL_ERROR "Bound installed OPAE prerequisites are absent")
endif()
add_executable(ahls_memory_inspect ahls_memory_inspect.c ahls_qualification_core.c)
target_include_directories(ahls_memory_inspect PRIVATE "${OPAE_INCLUDE_DIR}")
target_compile_options(ahls_memory_inspect PRIVATE -O2 -Wall -Wextra -Werror -pedantic)
target_link_libraries(ahls_memory_inspect PRIVATE "${OPAE_LIBRARY}")
# Deliberately no tests, custom commands, install or executable invocation.
